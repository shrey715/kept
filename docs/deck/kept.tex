% ---------------------------------------------------------------- the problem
\begin{frame}
  \kicker{The problem}\frametitle{Collections today asks for money on the wrong day}
  \source{evidence.md §1--2; tonight's nba.cards}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}\setlength\itemsep{0.3em}\small
        \item \textbf{Five systems, five keys per person.} Each files the same customer differently, so nobody sees
          the whole person.
        \item \textbf{Promises are dated by the calendar, not by pay.} A promise due before payday breaks; the bank's own
          data shows promised money paid is \textbf{+6.1 pts} higher when the due date falls just after payday.
        \item \textbf{More contact doesn't help.} The bank's randomised champion, two challengers and a hold-out keep
          promises at the same rate.
        \item \textbf{Hardship gets missed} in notes and calls, so vulnerable customers are pushed like anyone.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \kcard{\bignum{\KopenCases}{open cases tonight}\\[0.6em]\bignum{\KbalanceAtRisk}{balance at risk}\\[0.6em]\bignum{\KredCases}{need a hardship specialist}}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \kicker{Decisions, and the alternative we rejected}\frametitle{Why it's built this way}
  \source{docs/reports/evidence.md, extraction\_gate.md; the portfolio's A/B panel}
  \scriptsize
  \begin{tabular}{@{}L{3.3cm}L{4.0cm}L{6.0cm}@{}}
    \toprule
    \textbf{We chose} & \textbf{Instead of} & \textbf{Because} \\ \midrule
    Uplift per action & a propensity-to-pay score & propensity contacts people who'd pay anyway; uplift finds who a contact changes \\
    Timing (payday) & more channels or intensity & the bank's A/B: strategies keep promises at the same rate; payday timing adds +6.1 pts \\
    Local model for customer text & a hosted model for everything & customer words never leave the bank; no per-call cost; no quota mid-demo \\
    Certified metrics first & free-form SQL from a model & reviewed definitions answer most questions with no model-written SQL \\
    A graph with checks (LangGraph) & one prompt & each step is testable, traced, and fixable without retuning a prompt \\
    Two people for Red & one approver or the model & hardship decisions are never one person's or a model's \\
    Schema-constrained decoding & parse-and-retry & malformed replies 152 $\to$ 0 of 1,172 calls \\
    \bottomrule
  \end{tabular}
\end{frame}
