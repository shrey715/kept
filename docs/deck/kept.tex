\begin{frame}
  \kicker{The problem}\frametitle{Collections today asks for money on the wrong day}
  \source{evidence.md §1--2; tonight's nba.cards}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}\setlength\itemsep{0.3em}\small
        \item \textbf{Five systems, five keys per person.} Each files the same customer differently, so nobody sees
          the whole person.
        \item \textbf{Promises are dated by the calendar, not by pay.} A promise due before payday breaks; the bank's own
          data shows promised money paid is \textbf{\KpayMoney\ pts} higher when the due date falls within two days of payday.
        \item \textbf{More contact doesn't help.} The bank's randomised champion, two challengers and a hold-out keep
          promises at the same rate.
        \item \textbf{Hardship gets missed} in notes and calls, so vulnerable customers are pushed like anyone.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \kcard{\bignum{\KopenCases}{open cases tonight}\\[0.6em]\bignum{\KbalanceAtRisk}{balance at risk}\\[0.6em]\bignum{\KredCases}{need a hardship specialist}}
    \end{column}
  \end{columns}
\end{frame}
